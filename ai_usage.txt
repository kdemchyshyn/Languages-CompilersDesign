    peek and eat
    The parser owns the tokens and an index of the current one. Two primitives do all the reading: peek() returns the current token without moving (or nothing if there is none), eat() returns it and moves the index on. Everything else is one method per grammar rule, parse_program, parse_statement, parse_decl, …, and the right side of the rule dictates the body:
    •
    a sequence ("i32" [ "mut" ] ident ...) — eat the parts in order, complain when the current token is not the one the rule expects;
    •
    alternatives (decl | assign) — look at peek() and pick: i32 starts a declaration, a name starts an assignment. One token of look-ahead decides every choice in this language;
    •
    [ x ] — an if on peek(); { x } — a while on peek().
    If your lexer returns one vector of tokens per line, keep it: parse_program moves from line to line, and every other method works inside one line, where peek() returning nothing means “end of statement”. After a statement is parsed, the line must be empty — a leftover token is an error. If your lexer returns one flat vector, add an end-of-line token and check for it in the same place; both designs work.
    The tree
    The grammar dictates the classes as well as the functions. Where a rule says “either / or”, there is an abstract base with one subclass per alternative; where it says “this, then that”, there is a class whose fields are the parts that carry information. Quoted words disappear or become flags: mut becomes a boolean in the declaration, the braces and := leave nothing behind — the tree already says what they said.

    class Parser:
        def __init__(self, lines):

            self.lines = lines
            self.toks = []
            self.pos = 0
            self.current_line_num = 1
            self.last_col = 1

        def error(self, message):
            """Raises a CompileError with the line and column of the error."""
            if self.pos < len(self.toks):
                tok = self.toks[self.pos]
                line, col = tok.line, tok.col
            else:
                line = self.current_line_num
                col = self.last_col
            CompileError(f"line {line}:{col}: {message}")

        def peek(self):
            return self.toks[self.pos] if self.pos < len(self.toks) else None
        
        def eat(self):
            tok = self.toks[self.pos]
            self.pos += 1
            # Update last_col dynamically so end-of-line errors point right after the last token
            self.last_col = tok.col + len(tok.text)
            return tok
        
        def expect(self, kind, what):
            tok = self.peek()
            if tok is None:
                self.error(f"expected {what}, found end of line")
            if tok.kind != kind:
                self.error(f"expected {what}, got '{tok.text}'")
            return self.eat()
        
        def parse_program(self):
            stmts = []
            exit_node = None

            for toks in self.lines:
                if not toks: 
                    continue # Skip pure blank lines

                self.toks, self.pos = toks, 0
                self.current_line_num = toks[0].line
                # Default end of line column fallback if line fails immediately
                self.last_col = toks[-1].col + len(toks[-1].text) 
                
                if exit_node is not None:
                    self.error("statement after exit")

                tok = self.peek()
                if tok.kind == "exit":
                    exit_node = self.parse_exitStmt()
                else:
                    stmts.append(self.parse_statement())

                if self.peek() is not None:
                    self.error(f"unexpected '{self.peek().text}' after the statement")

            if exit_node is None:
                CompileError(f"line {self.current_line_num}:{self.last_col}: program without exit")

            start_line = stmts[0].line if stmts else exit_node.line
            start_col = stmts[0].col if stmts else exit_node.col
            return ProgramNode(start_line, start_col, stmts, exit_node)

        def parse_statement(self):
            tok = self.peek()
            if tok is None:
                self.error("expected a statement, found end of line")
            
            if tok.kind == "i32":
                return self.parse_decl()
            elif tok.kind == "ident":

    Here is the `CodeGen` visitor class that traverses the AST and generates the LLVM IR. It maintains a symbol table for variables and performs the required semantic checks (declaration before use, no redeclarations, and mutability before assignment).

    ```python
    from llvmlite import ir
    from errors import CompileError

    class CodeGen:
        def __init__(self, builder, printf_func, fmt_str):
            self.builder = builder
            self.printf = printf_func
            self.fmt = fmt_str
            self.env = {}  # Maps name -> {"ptr": ptr, "mutable": bool}

        def error(self, node, message):
            raise CompileError(f"line {node.line}:{node.col}: {message}")

        def visit_program(self, node):
            for stmt in node.statements:
                stmt.accept(self)
            node.exit.accept(self)

        def visit_decl(self, node):
            if node.name in self.env:
                self.error(node, f"variable '{node.name}' is already declared")
                
            init_val = node.init.accept(self)
            
            # Allocate stack space and store the initial value
            ptr = self.builder.alloca(ir.IntType(32), name=node.name)
            self.builder.store(init_val, ptr)
            
            # Save to symbol table
            self.env[node.name] = {"ptr": ptr, "mutable": node.mutable}

        def visit_assign(self, node):
            if node.name not in self.env:
                self.error(node, f"variable '{node.name}' is not declared")
                
            var_info = self.env[node.name]
            if not var_info["mutable"]:
                self.error(node, f"cannot assign to constant variable '{node.name}'")
                
            val = node.value.accept(self)
            self.builder.store(val, var_info["ptr"])

        def visit_exit(self, node):
            val = node.value.accept(self)
            
            # Cast the format string array to an i8* pointer for printf
            fmt_ptr = self.builder.bitcast(self.fmt, ir.PointerType(ir.IntType(8)))
            self.builder.call(self.printf, [fmt_ptr, val])
            
            # Return 0 from main
            self.builder.ret(ir.Constant(ir.IntType(32), 0))

        def visit_binop(self, node):
            left = node.left.accept(self)
            right = node.right.accept(self)
            
            if node.op == "+":
                return self.builder.add(left, right, name="addtmp")
            elif node.op == "-":
                return self.builder.sub(left, right, name="subtmp")
            elif node.op == "*":
                return self.builder.mul(left, right, name="multmp")
            else:
                self.error(node, f"unknown operator '{node.op}'")

        def visit_var(self, node):
            if node.name not in self.env:
                self.error(node, f"variable '{node.name}' is not declared")
                
            ptr = self.env[node.name]["ptr"]
            return self.builder.load(ptr, name=node.name)

        def visit_const(self, node):
            return ir.Constant(ir.IntType(32), node.value)

    ```
